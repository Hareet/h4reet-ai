  const words = hudWords(roomIndex());
  console.log(`P7NAMES ${JSON.stringify(words.names)}`);
  const widths = new Map<number, string>();
  for (const screen of sweepScreens()) {
    if (layoutFor(screen) !== "dock") continue;
    const m = hudModel(screen, words);
    const k = `${m.listRows}/${listRows(screen)}`;

# Split a diff -ruN patch into per-file bodies (the "diff ..." line dropped, since its flags differ between builds).
import sys, json
def files(p):
    out, cur, name = {}, [], None
    for line in open(p, encoding='utf-8', errors='surrogateescape'):
        if line.startswith('diff '):
            if name: out[name] = ''.join(cur)
            name, cur = line.rsplit(' ', 1)[-1].strip()[2:], []
        else:
            cur.append(line)
    if name: out[name] = ''.join(cur)
    return out
