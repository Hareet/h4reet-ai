function measure(offset: number[]): string {

  const { scene: part, index } = compose(tree, policy, LENIENT);

    const view = [0, 0, stage.w, stage.h] as const;

    const b = [...listOnly(areas, view, sideWall)].sort().join(",");

    }

  const reached = new Set<string>();

    for (let z = floor.bounds.z0; z <= floor.bounds.z1; z += 0.02) {

      for (let h = -180; h < 180; h += 15) {
