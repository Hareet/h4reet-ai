```python
#!/usr/bin/env python3
"""US6's merge recipe for docs/how-we-build/lifecycle.md, on a merge of feat/000-us6-docs into a main that holds the
us1 stack and US3.

The one conflict is in the enforcement table: the stack (T183's reviews) extended the row "The files prod reads and
the code the Author runs are human-only (ADR 0017)", and US6 rewrote the rows right after it. Keep the stack's row,
then US6's rows.
Usage: resolve_us6_sim.py <conflicted lifecycle.md>  (rewrites it in place)
"""

import re
import sys

ROW = "| The files prod reads and the code the Author runs are human-only (ADR 0017) |"

path = sys.argv[1]
with open(path, encoding="utf-8") as f:
    text = f.read()
blocks = list(
    re.finditer(
        r"^<<<<<<< [^\n]*\n(.*?)^=======\n(.*?)^>>>>>>> [^\n]*\n",
        text,
        re.DOTALL | re.MULTILINE,
    )
)
if len(blocks) != 1:
    sys.exit(f"expected one conflict in lifecycle.md, found {len(blocks)}")
ours = blocks[0].group(1).splitlines(keepends=True)
theirs = blocks[0].group(2).splitlines(keepends=True)
if len(ours) != 1 or not ours[0].startswith(ROW):
    sys.exit("the stack's side isn't the one extended row")
if not theirs or not theirs[0].startswith(ROW):
    sys.exit("US6's side doesn't start with that row")
if not ours[0].startswith(theirs[0].rstrip("\n").rstrip(" |")[:200]):
    sys.exit("US6 changed the row itself, not only the rows after it")
resolved = ours[0] + "".join(theirs[1:])
text = text[: blocks[0].start()] + resolved + text[blocks[0].end() :]
with open(path, "w", encoding="utf-8") as f:
    f.write(text)
print("lifecycle.md resolved: the stack's row, then US6's", len(theirs) - 1, "rows")
```
